Create file: test_db.py

import psycopg2
import os
from dotenv import load_dotenv

load_dotenv()

try:
    conn = psycopg2.connect(
        host=os.getenv('DB_HOST'),
        port=os.getenv('DB_PORT'),
        database=os.getenv('DB_NAME'),
        user=os.getenv('DB_USER'),
        password=os.getenv('DB_PASSWORD')
    )
    print("✅ Successfully connected to PostgreSQL!")

    cursor = conn.cursor()
    cursor.execute("SELECT table_name FROM information_schema.tables WHERE table_schema='public';")
    tables = cursor.fetchall()
    print(f"✅ Found {len(tables)} tables:")
    for table in tables:
        print(f"   - {table[0]}")

    cursor.close()
    conn.close()
except Exception as e:
    print(f"❌ Connection error: {e}")

📊 DATABASE SCHEMA
================================================================================

9 Tables Created:

1. STUDENTS
   Purpose: Store student information
   Columns: id, name, roll_number, class, email, phone, created_at, updated_at
   Constraints: roll_number is UNIQUE

2. CLASSES
   Purpose: Store class levels (I, II, III, ... VIII)
   Columns: id, name, description, created_at
   Constraints: name is UNIQUE

3. SUBJECTS
   Purpose: Subjects for each class (English, Math, Science)
   Columns: id, name, class_id, created_at
   Constraints: Foreign Key to classes, (name, class_id) UNIQUE

4. SKILLS
   Purpose: Skills under each subject
   Columns: id, name, description, subject_id, class_id, created_at
   Constraints: Foreign Keys, (name, subject_id) UNIQUE

5. GAMES
   Purpose: Games linked to skills
   Columns: id, name, description, skill_id, game_type, file_path, max_score, questions_count
   Constraints: Foreign Key to skills, name is UNIQUE

6. ASSIGNMENTS
   Purpose: Track game assignments to students
   Columns: id, student_id, game_id, class_id, subject_id, skill_id, assigned_by, status, deadline
   Constraints: Foreign Keys to all related tables

7. SCORES
   Purpose: Store scores when students complete games
   Columns: id, assignment_id, student_id, game_id, score, max_score, correct_answers, wrong_answers
   Constraints: Foreign Keys

8. AUDIO_FEEDBACK
   Purpose: Log audio feedback given to students
   Columns: id, score_id, feedback_type, frequency, duration
   Constraints: Foreign Key to scores

9. ANALYTICS
   Purpose: Aggregate student performance data
   Columns: id, student_id, class_id, skill_id, games_played, total_score, average_score
   Constraints: Foreign Keys

INDEXES: Created on frequently queried columns for performance
  - students(class)
  - assignments(student_id, game_id, status)
  - scores(student_id, game_id)
  - analytics(student_id, class_id)
